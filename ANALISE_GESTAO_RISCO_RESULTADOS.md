    # Remover todas as tags HTML
    texto_limpo = re.sub(r'<[^>]+>', '', texto)
    
    # Remover caracteres problemáticos
    texto_limpo = texto_limpo.replace('&', '&amp;')
    texto_limpo = texto_limpo.replace('<', '&lt;')
    texto_limpo = texto_limpo.replace('>', '&gt;')
    
    return texto_limpo
